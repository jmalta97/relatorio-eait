5.3 Etapa 3 — Gráficos com ggplot2

Código 
library(ggplot2)
library(scales)
# Definição da paleta de cores para os 4 tipos de obra
paleta_tipos <- c("Comercial" = "#0072B2", 
                  "Industrial" = "#E69F00", 
                  "Infraestrutura" = "#D55E00", 
                  "Residencial" = "#009E73")

# Tema padrão caso o tema_relatorio não esteja criado

g_q10 <- ggplot(dados, aes(area_construida, custo_total, color = tipo_obra)) + geom_point(size = 2.3, alpha = 0.75) + geom_smooth(method = "lm", se = FALSE) + scale_color_manual(values = paleta_tipos) + scale_y_continuous(labels = label_currency(prefix = "R$ ", big.mark = ".", decimal.mark = ",")) + labs(title = "Área e custo por tipo", subtitle = "Tendências lineares por tipologia", x = "Área (m²)", y = "Custo", color = "Tipo") + tema_relatorio
